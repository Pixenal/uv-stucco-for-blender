cmake_minimum_required(VERSION 3.2)
cmake_policy(SET CMP0091 NEW)
project(RuvmBlender)
add_library(RuvmBlender SHARED Src/RuvmBlender.c)
option(FIND_ZLIB "Try to find zlib" ON)
option(FIND_ZLIB_CONAN "Use conan to find zlib" OFF)
option(ZLIB_INCLUDE "Include directory for zlib" "")
option(ZLIB_LIB "Lib directory for zlib" "")
if (NOT FIND_ZLIB)
	if (NOT ZLIB_INCLUDE OR NOT ZLIB_LIB)
		message(FATAL_ERROR "FIND_ZLIB disabled, specify dir with ZLIB_LIB & ZLIB_INCLUDE")
	endif()
endif()
get_property(IS_MULTI_CONFIG GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if (NOT IS_MULTI_CONFIG)
	message("Building for single build type")
	if (NOT CMAKE_BUILD_TYPE)
		message("No build type specified, building Release")
		set(CMAKE_BUILD_TYPE Release "" FORCE)
	endif()
endif()
if (UNIX)
	message("Building for Unix")
	set(CMAKE_C_FLAGS "-D PLATFORM_LINUX")
	set(RELEASE_FLAGS "-ftree-vectorize -mavx2 -finline-functions -flto -O2 -pthread -D PLATFORM_LINUX -fPIC")
	set(DEBUG_FLAGS "-g -O0 -DEBUG -pthread -Wshadow -Wunused -Wunused-parameter -D PLATFORM_LINUX -fPIC")
	if (APPLE)
		set(RUVM_LIB_PATH ../RUVM/build/macosShared)
	else()
		
		set(RUVM_LIB_PATH ../RUVM/Build/Debug)
	endif()
	if (NOT CMAKE_BUILD_TYPE)
		message("No build type specified, building release")
		set(CMAKE_BUILD_TYPE "RELEASE")
		set(CMAKE_C_FLAGS_RELEASE ${RELEASE_C_FLAGS})
		message("Build type is " + CMAKE_BUILD_TYPE)
	endif()
	if(CMAKE_BUILD_TYPE MATCHES DEBUG)
		message("Debug build -- using gcc")
		set(CMAKE_C_FLAGS ${DEBUG_FLAGS})
	endif(CMAKE_BUILD_TYPE MATCHES DEBUG)
	if(CMAKE_BUILD_TYPE MATCHES RELEASE)
		set(CMAKE_C_COMPILER clang)
		set(CMAKE_C_FLAGS_RELEASE ${RELEASE_FLAGS})
	endif(CMAKE_BUILD_TYPE MATCHES RELEASE)
endif()
if (WIN32)
	message("Building for Windows")
	set(CMAKE_C_FLAGS "/D PLATFORM_WINDOWS /D WIN32 /D WINDOWS")
	set(CMAKE_C_STANDARD 11)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
target_include_directories(RuvmBlender PUBLIC Src)

target_include_directories(RuvmBlender PRIVATE Extern/Ruvm/Include)
add_subdirectory(Extern/Ruvm)

message("FIND_ZLIB is " + ${FIND_ZLIB})
if (FIND_ZLIB)
	message("Attempting to find zlib")
	find_package(ZLIB REQUIRED)
	message("--zlib include is " + ${ZLIB_INCLUDE_DIRS})
	message("Note that if the above are mismatched, there may be errors at compile time.")
	message ("Consider disabling FIND_ZLIB and specifying manually using ZLIB_LIB and ZLIB_INCLUDE. Or using Conan.")
	message("See github docs for more info.")
	target_include_directories(RuvmBlender PRIVATE ${ZLIB_INCLUDE_DIRS})
	target_link_libraries(RuvmBlender ${ZLIB_LIBRARIES} ${ZLIB_DEPENDENCIES} RUVM)
else()
	target_include_directories(RuvmBlender PRIVATE ${ZLIB_INCLUDE})
	target_link_libraries(RuvmBlender ${ZLIB_LIB} RUVM)
endif()
